// ═══════════════════════════════════════════════════ // Recruit Dashboard — Google Apps Script API // 貼到 Apps Script 後，部署為 Web App // 執行身分：我（你的 Google 帳號） // 存取權限：任何人 // ═══════════════════════════════════════════════════ const SHEET_ID = 'YOUR_GOOGLE_SHEET_ID'; // ← 換成你的 Sheet ID（網址列中間那串） // ── 允許 CORS，讓前端 Dashboard 可以呼叫 ── function doGet(e) { const action = e.parameter.action; let result; try { if (action === 'getTasks') result = getTasks(e.parameter); else if (action === 'getOKR') result = getOKR(e.parameter); else if (action === 'getVac') result = getVacancies(); else if (action === 'getAll') result = getAllData(); else result = { error: 'Unknown action' }; } catch (err) { result = { error: err.message }; } return ContentService .createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { let body; try { body = JSON.parse(e.postData.contents); } catch (err) { return jsonResponse({ error: 'Invalid JSON' }); } // ── 權限驗證：只有寫入 token 符合才允許 ── // 在 Settings 表加一列：key=write_token, value=你自己設定的密碼 const token = body.token; const settings = getSettings(); if (token !== settings['write_token']) { return jsonResponse({ error: 'Unauthorized' }); } let result; try { const action = body.action; if (action === 'upsertTask') result = upsertTask(body.data); else if (action === 'deleteTask') result = deleteTask(body.data.task_id); else if (action === 'upsertVac') result = upsertVac(body.data); else if (action === 'deleteVac') result = deleteVac(body.data.vac_id); else if (action === 'updateOKR') result = updateOKR(body.data); else result = { error: 'Unknown action' }; } catch (err) { result = { error: err.message }; } return jsonResponse(result); } function jsonResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } // ════════════════════════════════ // READ 函式 // ════════════════════════════════ function getAllData() { return { tasks: getTasks({}), okr: getOKR({}), vacancies: getVacancies(), members: getMembers(), settings: getSettings(), }; } function getTasks(params) { const sheet = getSheet('Tasks'); const rows = sheetToObjects(sheet); // 可依 quarter / member 篩選 return rows.filter(r => { if (params.quarter && r.quarter !== params.quarter) return false; if (params.member && r.member !== params.member) return false; return true; }); } function getOKR(params) { const sheet = getSheet('OKR'); const rows = sheetToObjects(sheet); if (params.quarter) return rows.filter(r => r.quarter === params.quarter); return rows; } function getVacancies() { return sheetToObjects(getSheet('Vacancies')); } function getMembers() { return sheetToObjects(getSheet('Members')); } function getSettings() { const rows = sheetToObjects(getSheet('Settings')); const obj = {}; rows.forEach(r => { obj[r.key] = r.value; }); return obj; } // ════════════════════════════════ // WRITE 函式 // ════════════════════════════════ // Tasks：members 只能更新 actual_hours / completion_pct / status / note // Lead 可以更新所有欄位 // role 由前端傳入（'lead' or 'member'），token 已驗過 function upsertTask(data) { const sheet = getSheet('Tasks'); const headers = getHeaders(sheet); const rows = sheet.getDataRange().getValues(); // 自動補 updated_at data.updated_at = new Date().toISOString(); // 狀態聯動：完成 → pct=100，未開始 → pct=0 if (data.status === '完成') data.completion_pct = 100; if (data.status === '未開始') data.completion_pct = 0; // 找是否已有此 task_id const idIdx = headers.indexOf('task_id'); for (let i = 1; i < rows.length; i++) { if (String(rows[i][idIdx]) === String(data.task_id)) { // 更新既有列 const role = data._role || 'member'; const allowedForMember = ['actual_hours','completion_pct','status','note','updated_at']; headers.forEach((h, col) => { if (h in data) { if (role === 'lead' || allowedForMember.includes(h)) { sheet.getRange(i + 1, col + 1).setValue(data[h]); } } }); return { success: true, action: 'updated', task_id: data.task_id }; } } // 新增列（只有 lead 可以） if (data._role === 'member') return { error: 'Members cannot create new tasks' }; if (!data.task_id) data.task_id = 'T' + Date.now(); const newRow = headers.map(h => (h in data ? data[h] : '')); sheet.appendRow(newRow); return { success: true, action: 'created', task_id: data.task_id }; } function deleteTask(task_id) { return deleteRowById(getSheet('Tasks'), 'task_id', task_id); } function upsertVac(data) { const sheet = getSheet('Vacancies'); const headers = getHeaders(sheet); const rows = sheet.getDataRange().getValues(); data.updated_at = new Date().toISOString(); const idIdx = headers.indexOf('vac_id'); for (let i = 1; i < rows.length; i++) { if (String(rows[i][idIdx]) === String(data.vac_id)) { headers.forEach((h, col) => { if (h in data) sheet.getRange(i + 1, col + 1).setValue(data[h]); }); return { success: true, action: 'updated', vac_id: data.vac_id }; } } if (!data.vac_id) data.vac_id = 'V' + Date.now(); sheet.appendRow(headers.map(h => data[h] || '')); return { success: true, action: 'created', vac_id: data.vac_id }; } function deleteVac(vac_id) { return deleteRowById(getSheet('Vacancies'), 'vac_id', vac_id); } // OKR 整張覆寫（結構較複雜，用整季替換較安全） function updateOKR(data) { // data = { quarter: 'Q3', rows: [...] } const sheet = getSheet('OKR'); const headers = getHeaders(sheet); const rows = sheet.getDataRange().getValues(); // 刪除該季所有列（從後往前刪避免 index 錯位） const qIdx = headers.indexOf('quarter'); for (let i = rows.length - 1; i >= 1; i--) { if (rows[i][qIdx] === data.quarter) { sheet.deleteRow(i + 1); } } // 重新寫入 data.rows.forEach(row => { sheet.appendRow(headers.map(h => row[h] || '')); }); return { success: true, quarter: data.quarter, count: data.rows.length }; } // ════════════════════════════════ // 工具函式 // ════════════════════════════════ function getSheet(name) { const ss = SpreadsheetApp.openById(SHEET_ID); const sheet = ss.getSheetByName(name); if (!sheet) throw new Error('Sheet not found: ' + name); return sheet; } function getHeaders(sheet) { return sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(String); } function sheetToObjects(sheet) { const data = sheet.getDataRange().getValues(); if (data.length < 2) return []; const headers = data[0].map(String); return data.slice(1).map(row => { const obj = {}; headers.forEach((h, i) => { obj[h] = row[i]; }); return obj; }); } function deleteRowById(sheet, idField, idValue) { const headers = getHeaders(sheet); const rows = sheet.getDataRange().getValues(); const idIdx = headers.indexOf(idField); for (let i = rows.length - 1; i >= 1; i--) { if (String(rows[i][idIdx]) === String(idValue)) { sheet.deleteRow(i + 1); return { success: true, deleted: idValue }; } } return { error: 'Row not found' }; }
